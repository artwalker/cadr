function model<TApi extends Api>(
	api: TApi,
	provider: string,
	id: string,
	baseUrl: string,
	overrides: Partial<Model<TApi>> = {},
): Model<TApi> {
	return {
		id,
		name: id,
		api,
		provider,
		baseUrl,
		reasoning: false,
		input: ["text"],
		cost: {
			input: 0,
			output: 0,
			cacheRead: 0,
			cacheWrite: 0,
		},
		contextWindow: 1_048_576,
		maxTokens: 4096,
		...overrides,
	};
}

async function capturePayload<T, TOptions extends StreamOptions = StreamOptions>(
	start: (options: TOptions) => { result(): Promise<unknown> },
	options: TOptions = {} as TOptions,
): Promise<T> {
	let captured: T | undefined;
	const response = start({
		...options,
		apiKey: "payload-capture-key",
		maxRetries: 0,
		onPayload(payload) {
			captured = structuredClone(payload) as T;
			throw new Error("payload captured");
		},
	});
	await response.result();
	if (captured === undefined) {
		throw new Error("Provider did not invoke onPayload");
	}
	return captured;
}

function googleWireBody(params: {
	model: string;
	contents: unknown;
	config?: {
		maxOutputTokens?: number;
		systemInstruction?: string | object;
		tools?: unknown;
		toolConfig?: unknown;
		thinkingConfig?: unknown;
	};
}): Record<string, unknown> {
	const config = params.config ?? {};
	const body: Record<string, unknown> = {
		contents: params.contents,
	};
	if (config.systemInstruction !== undefined) {
		body.systemInstruction =
			typeof config.systemInstruction === "string"
				? { parts: [{ text: config.systemInstruction }] }
				: config.systemInstruction;
	}
	if (config.maxOutputTokens !== undefined) {
		body.generationConfig = {
			maxOutputTokens: config.maxOutputTokens,
		};
	}
	if (config.tools !== undefined) {
		body.tools = config.tools;
	}
	if (config.toolConfig !== undefined) {
		body.toolConfig = config.toolConfig;
	}
	if (config.thinkingConfig !== undefined) {
		body.thinkingConfig = config.thinkingConfig;
	}
	return body;
}

function writeFixture(name: string, payload: unknown): void {
	fs.mkdirSync(outputDirectory, { recursive: true });
	fs.writeFileSync(
		path.join(outputDirectory, name),
		`${JSON.stringify(payload, null, 2)}\n`,
		"utf8",
	);
}

function requireModel<TApi extends Api>(
	models: Record<string, Model<TApi>>,
	id: string,
): Model<TApi> {
	const selected = models[id];
	if (selected === undefined) {
		throw new Error(`Missing fixed catalog model: ${id}`);
	}
	return selected;
}

function readCatalog<TApi extends Api>(
	provider: string,
): Record<string, Model<TApi>> {
	const file = path.join(catalogDirectory, "providers", `${provider}.json`);
	const parsed: unknown = JSON.parse(fs.readFileSync(file, "utf8"));
	if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
		throw new Error(`Invalid generated catalog: ${file}`);
	}
	return parsed as Record<string, Model<TApi>>;
}
